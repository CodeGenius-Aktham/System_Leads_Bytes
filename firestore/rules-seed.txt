// ==========================================================================
// REGLAS TEMPORALES — SOLO PARA SEMBRAR `leadOwners`
//
// Permiten que una cuenta con sesión escriba la colección de asignaciones.
// Publicar, correr firestore/seed-leadOwners.js, y reemplazar de inmediato
// por firestore/rules.txt.
//
// No dejar estas reglas publicadas: con sesión, cualquiera podría
// reasignarse los leads de otro.
// ==========================================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /leadOwners/{leadId} {
      allow read, write: if request.auth != null;
    }

    match /leadStatus/{leadId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && request.resource.data.keys().hasOnly(
                        ['status', 'updatedAt', 'updatedBy', 'ts'])
                   && request.resource.data.status in
                        ['sin_contactar', 'contactado', 'cliente', 'rechazado']
                   && request.resource.data.ts is int;
    }

    match /{document=**} { allow read, write: if false; }
  }
}
